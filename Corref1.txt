var validStatuses = new List<string> { "Active", "Pending", "Completed" };

// Convert list to a SQL-compatible format
string sqlQuery = "SELECT * FROM [Tracking.MessageLog] WHERE [StatusType] IN ({0})";

// Create parameterized query dynamically
var parameters = validStatuses
    .Select((status, index) => new SqlParameter($"@p{index}", status))
    .ToArray();

string inClause = string.Join(", ", parameters.Select(p => p.ParameterName));

var results = await dbContext.TrackingMessageLogs
    .FromSqlRaw(string.Format(sqlQuery, inClause), parameters)
    .ToListAsync();
