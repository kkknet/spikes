@Component({
  selector: 'app-payment-transactions',
  standalone: true,

  @ViewChild('modalContainer') modalContainer!: ElementRef<HTMLElement>;

  private apiService = inject(ApiService);
  private fb         = inject(FormBuilder);
  scoapMaxLength     = 14;

  agencyCodes: string[]     = [];
  programCatCodes: string[] = [];
  collectionTypes: string[] = [];

  lookupOptions = signal<Record<string, unknown>>({});

  readonly yesNoOptions     = ['Y', 'N'];
  readonly availTypeOptions = ['X', 'F', ' '];

  // 9 years from now
  readonly yearOptions = Array.from({ length: 10 }, (_, i) =>
    (new Date().getFullYear() + i).toString()
  );

  rows: BfmField[][] = [];

  @Input() stepWizard!: StepWizardComponent;
  form: FormGroup = this.fb.group({});
  isLoading       = signal(true);

  // ── Single source of truth for txn rows ────────────────────────────────────
  private readonly txnCodes = [
    '105', '107', '109', '115', '117',
    '118', '171', '172', '173', '174',
  ];

  private readonly defaultBetc: Record<string, string> = {
    '105': 'DISGF',   '107': 'DISGFAJ', '109': 'DISGF',
    '115': 'DISGFAJ', '117': 'DISGFAJ', '118': '',
    '171': 'DISGFAJ', '172': 'DISGFAJ', '173': 'DISGFAJ', '174': 'DISGFAJ',
  };

  private readonly defaultRevBetc: Record<string, string> = {
    '105': 'DISGFAJ', '107': 'DISGF',   '109': 'DISGFAJ',
    '115': 'DISGF',   '117': 'DISGF',   '118': '',
    '171': 'DISGF',   '172': 'DISGF',   '173': 'DISGF',   '174': 'DISGF',
  };

  private readonly selectedByDefault = [
    '105', '107', '109', '115', '117', '171', '172',

  // ───────────────────────────────────────────────────────────────────────────

  ngOnInit() {
    this.initializeData();
  }

  ngAfterViewInit() {}

  private initializeData() {
    const lookupCodes = [
      'ACA', 'MAC', 'FAL', 'FCL', 'FPL',
      'IRC', 'CTU', 'TOF', 'SAC', 'ALC', 'DPC',
      'AUTO_IND', 'BAL_ACTION', 'CTL_TOT', 'BETC', 'REV_BETC', 'SUB_ACCT',
    ];

    const lookupObservables = lookupCodes.map((code) =>
      this.fetchLookup(code)
    );

    // PR Fix: Use forkJoin to handle everything in one stream
    forkJoin({
      lookups:           forkJoin(lookupObservables),
      agencyCodes:       this.fetchAgencyCodes(),
      programCategories: this.fetchProgramCategories(),
      collectionTypes:   this.fetchCollectionTypes(),
    })
      .pipe(
        finalize(() => {
          this.buildForm();         // MUST be before isLoading.set(false)
          this.isLoading.set(false);
        })
      )
      .subscribe({
        next: (results) => {
          const mapped: Record<string, unknown> = {};

          lookupCodes.forEach((code, i) => {
            const raw = results.lookups[i];

            // Handle both raw array and wrapped API response shapes
            if (Array.isArray(raw)) {
              mapped[code] = raw;
            } else if (raw && typeof raw === 'object') {
              const obj = raw as Record<string, unknown>;
              mapped[code] =
                (obj['data']   as string[]) ??
                (obj['items']  as string[]) ??
                (obj['value']  as string[]) ??
                [];
            } else {
              mapped[code] = [];
            }
          });

          this.lookupOptions.set(mapped);

          // Debug — remove once confirmed working
          console.log('lookupOptions BETC:',     this.lookupOptions()['BETC']);
          console.log('lookupOptions AUTO_IND:', this.lookupOptions()['AUTO_IND']);

          this.agencyCodes     = this.extractArray(results.agencyCodes);
          this.programCatCodes = this.extractArray(results.programCategories);
          this.collectionTypes = this.extractArray(results.collectionTypes);
        },
        error: (err) => console.error('Data initialization failed', err),
      });
  }

  // ── Safely extracts string[] from any API response shape ───────────────────
  private extractArray(raw: unknown): string[] {
    if (Array.isArray(raw)) return raw as string[];
    if (raw && typeof raw === 'object') {
      const obj = raw as Record<string, unknown>;
      if (Array.isArray(obj['data']))   return obj['data']  as string[];
      if (Array.isArray(obj['items']))  return obj['items'] as string[];
      if (Array.isArray(obj['value']))  return obj['value'] as string[];
    }
    return [];
  }

  // ── Returns a safe string[] for *ngFor — handles any API response shape ────
  getOptions(lookupCode: string | undefined): string[] {
    if (!lookupCode) return [];
    const val = this.lookupOptions()[lookupCode];

    if (Array.isArray(val)) return val as string[];

    if (val && typeof val === 'object') {
      const obj = val as Record<string, unknown>;
      if (Array.isArray(obj['data']))   return obj['data']  as string[];
      if (Array.isArray(obj['items']))  return obj['items'] as string[];
      if (Array.isArray(obj['value']))  return obj['value'] as string[];
      // Last resort — return string values from the object
      return Object.values(obj).filter(
        (v) => typeof v === 'string'
      ) as string[];
    }
    return [];
  }

  // ── Replace these with real apiService calls ────────────────────────────────
  private fetchLookup(code: string): Observable<unknown> {
    return this.apiService.getLookup(code).pipe(
      catchError(() => of([]))
    );
  }

  private fetchAgencyCodes(): Observable<unknown> {
    return this.apiService.getAgencyCodes().pipe(
      catchError(() => of([]))
    );
  }

  private fetchProgramCategories(): Observable<unknown> {
    return this.apiService.getProgramCategories().pipe(
      catchError(() => of([]))
    );
  }

  private fetchCollectionTypes(): Observable<unknown> {
    return this.apiService.getCollectionTypes().pipe(
      catchError(() => of([]))
    );
  }
  // ───────────────────────────────────────────────────────────────────────────

  private buildForm() {
    const group: Record<string, unknown> = {};

    this.txnCodes.forEach((code, i) => {
      const isSelected = this.selectedByDefault.includes(code);
      group[`pt_select_${i}`]    = [isSelected];
      group[`pt_autoInd_${i}`]   = [''];
      group[`pt_balAction_${i}`] = [''];
      group[`pt_ctlTot_${i}`]    = [''];
      group[`pt_betc_${i}`]      = [
        this.defaultBetc[code],
        isSelected ? [Validators.required] : [],
      ];
      group[`pt_revBetc_${i}`]   = [
        this.defaultRevBetc[code],
        isSelected ? [Validators.required] : [],
      ];
      group[`pt_subAcct_${i}`]   = ['000', [Validators.required]];
    });

    this.form = this.fb.group(group);
    this.buildRows();
  }

  private buildRows() {
    // Header row — all __spacer__ keys, never bound to form
    const headerRow: BfmField[] = [
      { key: '__spacer__', label: 'Select',     type: 'label' },
      { key: '__spacer__', label: 'Txn Code',   type: 'label' },
      { key: '__spacer__', label: 'Auto Ind',   type: 'label' },
      { key: '__spacer__', label: 'Bal Action', type: 'label' },
      { key: '__spacer__', label: 'Ctl Tot',    type: 'label' },
      { key: '__spacer__', label: 'BETC',       type: 'label' },
      { key: '__spacer__', label: 'REV BETC',   type: 'label' },
      { key: '__spacer__', label: 'Sub Acct',   type: 'label' },
    ];

    // One data row per txn code — keys MUST match buildForm() exactly
    const dataRows: BfmField[][] = this.txnCodes.map((code, i) => [
      { key: `pt_select_${i}`,    label: '',   type: 'checkbox'                                        },
      { key: '__spacer__',        label: code, type: 'label'                                           },
      { key: `pt_autoInd_${i}`,   label: '',   type: 'dropdown', lookupCode: 'AUTO_IND'                },
      { key: `pt_balAction_${i}`, label: '',   type: 'dropdown', lookupCode: 'BAL_ACTION'              },
      { key: `pt_ctlTot_${i}`,    label: '',   type: 'dropdown', lookupCode: 'CTL_TOT'                 },
      { key: `pt_betc_${i}`,      label: '',   type: 'dropdown', lookupCode: 'BETC',     required: true },
      { key: `pt_revBetc_${i}`,   label: '',   type: 'dropdown', lookupCode: 'REV_BETC', required: true },
      { key: `pt_subAcct_${i}`,   label: '',   type: 'dropdown', lookupCode: 'SUB_ACCT', required: true },
    ]);

    this.rows = [headerRow, ...dataRows];
    console.log('rows in pmnt txns', this.rows);
  }

  onSelectChange(index: number): void {
    const isSelected  = this.form.get(`pt_select_${index}`)?.value as boolean;
    const betcCtrl    = this.form.get(`pt_betc_${index}`);
    const revBetcCtrl = this.form.get(`pt_revBetc_${index}`);

    if (isSelected) {
      betcCtrl?.setValidators([Validators.required]);
      revBetcCtrl?.setValidators([Validators.required]);
    } else {
      betcCtrl?.clearValidators();
      revBetcCtrl?.clearValidators();
    }
    betcCtrl?.updateValueAndValidity();
    revBetcCtrl?.updateValueAndValidity();
  }

  isFieldRequired(field: BfmField): boolean {
    return field.required === true;
  }

  isReadonly(field: BfmField): boolean {
    return field.readonly === true;
  }

  onSubmit(): void {
    if (this.form.valid) {
      const payload = this.txnCodes.map((code, i) => ({
        txnCode:   code,
        selected:  this.form.get(`pt_select_${i}`)?.value    as boolean,
        autoInd:   this.form.get(`pt_autoInd_${i}`)?.value   as string,
        balAction: this.form.get(`pt_balAction_${i}`)?.value as string,
        ctlTot:    this.form.get(`pt_ctlTot_${i}`)?.value    as string,
        betc:      this.form.get(`pt_betc_${i}`)?.value      as string,
        revBetc:   this.form.get(`pt_revBetc_${i}`)?.value   as string,
        subAcct:   this.form.get(`pt_subAcct_${i}`)?.value   as string,
      }));
      console.log('Payment Txns payload:', payload);
      // Replace with: this.apiService.savePaymentTxns(payload).subscribe(...)
    } else {
      this.form.markAllAsTouched();
    }
  }

  onCancel(): void {
    this.buildForm();
  }
}
