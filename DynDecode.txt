using Microsoft.AspNetCore.Mvc.Filters;
using System;
using System.Collections.Generic;
using System.Text;

public class DecodeBase64QueryParamsAttribute : ActionFilterAttribute
{
    public override void OnActionExecuting(ActionExecutingContext context)
    {
        var queryParams = context.HttpContext.Request.Query;
        var decodedParams = new Dictionary<string, string>();

        foreach (var param in queryParams)
        {
            if (IsBase64String(param.Value))
            {
                decodedParams[param.Key] = DecodeBase64String(param.Value);
            }
            else
            {
                decodedParams[param.Key] = param.Value;
            }
        }

        // Here you can decide how to provide the decoded values to your action method.
        // For example, you could add them to the context's RouteData or Items collection.
        foreach (var decodedParam in decodedParams)
        {
            context.RouteData.Values[decodedParam.Key] = decodedParam.Value;
        }

        base.OnActionExecuting(context);
    }

    private bool IsBase64String(string s)
    {
        s = s.Trim();
        return (s.Length % 4 == 0) && Convert.TryFromBase64String(s, new Span<byte>(new byte[s.Length]), out _);
    }

    private string DecodeBase64String(string base64String)
    {
        byte[] data = Convert.FromBase64String(base64String);
        return Encoding.UTF8.GetString(data);
    }
}
