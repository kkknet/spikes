// ─── onSaveDraft ──────────────────────────────────────────────────────────

describe(‘onSaveDraft’, () => {
beforeEach(() => {
component.programYears = [
{ year: 2026, status: ‘Active’,   selected: true },
{ year: 2027, status: ‘Inactive’, selected: false }
];
});

```
it('should call post with the correct endpoint, payload, and options', () => {
  apiServiceSpy.post.mockReturnValue(of('OK'));

  component.onSaveDraft();

  expect(apiServiceSpy.post).toHaveBeenCalledWith(
    'pfrsuiapi/AccountingProgramYears/saveDraft',
    {
      programYears: [{ year: 2026, status: 'Active', selected: true }],
      status: 'DRAFT'
    },
    { responseType: 'text' }
  );
});

it('should set saveSubmitResultMessage to "Draft saved successfully" on success', () => {
  apiServiceSpy.post.mockReturnValue(of('OK'));

  component.onSaveDraft();

  expect(component.saveSubmitResultMessage).toBe('Draft saved successfully');
});

it('should clear saveSubmitResultMessage after 5 seconds', () => {
  jest.useFakeTimers();
  apiServiceSpy.post.mockReturnValue(of('OK'));

  component.onSaveDraft();
  expect(component.saveSubmitResultMessage).toBe('Draft saved successfully');

  jest.advanceTimersByTime(5000);

  expect(component.saveSubmitResultMessage).toBe('');
});

it('should not clear saveSubmitResultMessage before 5 seconds elapse', () => {
  jest.useFakeTimers();
  apiServiceSpy.post.mockReturnValue(of('OK'));

  component.onSaveDraft();
  jest.advanceTimersByTime(4999);

  expect(component.saveSubmitResultMessage).toBe('Draft saved successfully');
});

it('should set errorMessage and return early when no years are selected', () => {
  component.programYears = [
    { year: 2026, status: 'Inactive', selected: false }
  ];

  component.onSaveDraft();

  expect(component.errorMessage).toBe('Please select at least one year');
  expect(apiServiceSpy.post).not.toHaveBeenCalled();
});

it('should clear a previous errorMessage before posting', () => {
  component.errorMessage = 'Previous error';
  apiServiceSpy.post.mockReturnValue(of('OK'));

  component.onSaveDraft();

  expect(component.errorMessage).toBe('');
});

it('should set errorMessage on API error', () => {
  const err = new HttpErrorResponse({ status: 500, statusText: 'Server Error' });
  apiServiceSpy.post.mockReturnValue(throwError(() => err));

  component.onSaveDraft();

  expect(component.errorMessage).toBe('Error saving accounting program years draft');
});

it('should not call post when programYears list is empty', () => {
  component.programYears = [];

  component.onSaveDraft();

  expect(apiServiceSpy.post).not.toHaveBeenCalled();
});
```

// ─── End-to-end workflow ──────────────────────────────────────────────────

describe(‘End-to-end workflow’, () => {
it(‘should support full draft workflow: init → select all → save draft’, () => {
apiServiceSpy.post.mockReturnValue(of(‘OK’));

```
  component.ngOnInit();
  expect(component.programYears.length).toBe(10);

  component.selectAllYears();
  expect(component.isAnyYearSelected()).toBe(true);

  component.onSaveDraft();

  expect(component.saveSubmitResultMessage).toBe('Draft saved successfully');
  expect(component.errorMessage).toBe('');
});

it('should support full submit workflow: init → partial select → submit', () => {
  apiServiceSpy.post.mockReturnValue(of('OK'));

  component.ngOnInit();
  component.programYears[0].selected = true;
  component.onSubmit();

  expect(component.saveSubmitResultMessage).toBe('Years submitted successfully');
  expect(component.errorMessage).toBe('');
});

it('should reflect correct count after select all then deselect all', () => {
  component.ngOnInit();

  component.selectAllYears();
  expect(component.getSelectedCount()).toBe(10);

  component.deselectAllYears();
  expect(component.getSelectedCount()).toBe(0);
  expect(component.isAnyYearSelected()).toBe(false);
});

it('should reflect correct count after toggling individual years', () => {
  component.ngOnInit();
  const [first, second, third] = component.programYears;

  component.toggleYearSelection(first);
  component.toggleYearSelection(second);
  component.toggleYearSelection(third);
  expect(component.getSelectedCount()).toBe(3);

  component.toggleYearSelection(first);
  expect(component.getSelectedCount()).toBe(2);
});

it('should show error then allow save after selecting a year', () => {
  apiServiceSpy.post.mockReturnValue(of('OK'));

  component.ngOnInit();
  component.onSaveDraft();
  expect(component.errorMessage).toBe('Please select at least one year');

  component.programYears[0].selected = true;
  component.onSaveDraft();
  expect(component.saveSubmitResultMessage).toBe('Draft saved successfully');
});
```
