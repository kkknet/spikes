import * as fs from 'fs';
import * as crypto from 'crypto';

// Function to Base64URL encode a buffer
function base64urlEncode(buffer: Buffer): string {
  return buffer.toString('base64')
    .replace(/\+/g, '-')
    .replace(/\//g, '_')
    .replace(/=+$/, '');
}

// Read the certificate file (PEM format)
const certPem = fs.readFileSync('path/to/your/certificate.pem', 'utf8');

// Extract the Base64 content between the header and footer
const certBase64 = certPem
  .replace('-----BEGIN CERTIFICATE-----', '')
  .replace('-----END CERTIFICATE-----', '')
  .replace(/\r?\n|\r/g, ''); // Remove newlines

// Decode the Base64 to get DER bytes
const certDer = Buffer.from(certBase64, 'base64');

// Compute the SHA-1 hash of the DER bytes
const sha1Hash = crypto.createHash('sha1').update(certDer).digest();

// Base64URL-encode the hash to get x5t
const x5t = base64urlEncode(sha1Hash);

console.log('x5t:', x5t);







import { ConfidentialClientApplication } from "@azure/msal-node";
import * as fs from "fs";
import * as path from "path";

// Load your certificate and private key
const certPath = path.join(__dirname, "cert.pem");
const keyPath = path.join(__dirname, "key.pem");
const certificate = fs.readFileSync(certPath, "utf8");
const privateKey = fs.readFileSync(keyPath, "utf8");

// Initialize the MSAL client
const clientConfig = {
    auth: {
        clientId: "YOUR_CLIENT_ID",
        authority: "https://YOUR_TENANT_NAME.b2clogin.com/YOUR_TENANT_NAME.onmicrosoft.com/B2C_1A_SIGNUP_SIGNIN",
        clientCertificate: {
            thumbprint: "YOUR_CERT_THUMBPRINT",
            privateKey: privateKey,
            x5c: certificate
        }
    }
};

const cca = new ConfidentialClientApplication(clientConfig);

// Request JWT token
const tokenRequest = {
    scopes: ["https://YOUR_TENANT_NAME.onmicrosoft.com/api/read"],
};

async function getToken() {
    try {
        const response = await cca.acquireTokenByClientCredential(tokenRequest);
        console.log("Access Token:", response.accessToken);
    } catch (error) {
        console.error("Error acquiring token:", error);
    }
}

getToken();
