describe(‘initializeYears’, () => {
it(‘should generate 10 years starting from current year’, () => {
const currentYear = new Date().getFullYear();

```
  component.initializeYears();

  expect(component.programYears.length).toBe(10);
  expect(component.programYears[0].year).toBe(currentYear);
  expect(component.programYears[9].year).toBe(currentYear + 9);
});

it('should initialize all years with Inactive status and unselected', () => {
  component.initializeYears();

  component.programYears.forEach(year => {
    expect(year.status).toBe('Inactive');
    expect(year.selected).toBe(false);
  });
});

it('should generate consecutive years', () => {
  component.initializeYears();

  for (let i = 1; i < component.programYears.length; i++) {
    expect(component.programYears[i].year).toBe(component.programYears[i - 1].year + 1);
  }
});

it('should clear and regenerate years on multiple calls', () => {
  component.initializeYears();
  const firstCall = [...component.programYears];
  
  component.initializeYears();
  const secondCall = [...component.programYears];

  expect(firstCall.length).toBe(secondCall.length);
  expect(firstCall[0].year).toBe(secondCall[0].year);
});
```

describe(‘onSaveDraft’, () => {
beforeEach(() => {
component.programYears = [
{ year: 2026, status: ‘Active’, selected: true },
{ year: 2027, status: ‘Inactive’, selected: false }
];
});

```
it('should save draft successfully', () => {
  apiServiceSpy.post.and.returnValue(of('Draft saved successfully'));
  
  component.onSaveDraft();

  expect(apiServiceSpy.post).toHaveBeenCalledWith(
    'pfrsuiapi/AccountingProgramYears/saveDraft',
    jasmine.objectContaining({
      status: 'DRAFT',
      programYears: jasmine.any(Array)
    }),
    { responseType: 'text' }
  );
  expect(component.saveSubmitResultMessage).toBe('Draft saved successfully');
});

it('should show error when no years are selected', () => {
  component.programYears = [
    { year: 2026, status: 'Inactive', selected: false }
  ];

  component.onSaveDraft();

  expect(component.errorMessage).toBe('Please select at least one year');
  expect(apiServiceSpy.post).not.toHaveBeenCalled();
});

it('should clear error message before saving', () => {
  component.errorMessage = 'Previous error';
  apiServiceSpy.post.and.returnValue(of('Success'));

  component.onSaveDraft();

  expect(component.errorMessage).toBe('');
});

it('should handle save draft API error', () => {
  const errorResponse = new HttpErrorResponse({
    error: 'Save failed',
    status: 500,
    statusText: 'Internal Server Error'
  });

  apiServiceSpy.post.and.returnValue(throwError(() => errorResponse));

  component.onSaveDraft();

  expect(component.errorMessage).toBe('Error saving accounting program years draft');
});

it('should clear success message after 5 seconds', (done) => {
  jasmine.clock().install();
  apiServiceSpy.post.and.returnValue(of('Success'));

  component.onSaveDraft();

  expect(component.saveSubmitResultMessage).toBe('Draft saved successfully');

  jasmine.clock().tick(5001);

  expect(component.saveSubmitResultMessage).toBe('');
  jasmine.clock().uninstall();
  done();
});
```

});
