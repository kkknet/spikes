<div *ngIf="!isLoading() && form; else loadingTemplate">
  <div
    #modalContainer
    tabindex="-1"
    class="fsa-grid"
    [formGroup]="form"
  >
    <ng-container *ngFor="let row of rows">
      <ng-container *ngFor="let field of row">

        <div class="fsa-grid__1/12">

          <!-- SPACER — header labels and txn code display, never bound to form -->
          <ng-container *ngIf="field.key === '__spacer__'">
            <label class="fsa-field__label">{{ field.label }}</label>
          </ng-container>

          <!-- INPUT -->
          <ng-container *ngIf="field.type === 'input' && field.key !== '__spacer__'">
            <div *ngIf="isReadonly(field)" class="fsa-read-only-text">
              | {{ form.get(field.key)?.value }}
            </div>
            <input
              *ngIf="!isReadonly(field)"
              type="text"
              class="fsa-input"
              [formControlName]="field.key"
              [attr.maxlength]="field.maxLength || 40"
            />
          </ng-container>

          <!-- CHECKBOX -->
          <ng-container *ngIf="field.type === 'checkbox' && field.key !== '__spacer__'">
            <div class="fsa-field">
              <input
                class="fsa-checkbox fsa-checkbox--solo"
                [formControlName]="field.key"
                type="checkbox"
                [id]="'field_' + field.key"
                (change)="onSelectChange(+field.key.split('_')[2])"
              />
              <label
                class="fsa-field__label"
                [for]="'field_' + field.key"
              ></label>
            </div>
          </ng-container>

          <!-- DROPDOWN -->
          <ng-container *ngIf="field.type === 'dropdown' && field.key !== '__spacer__'">
            <select
              class="fsa-select"
              [formControlName]="field.key"
              [id]="'field_' + field.key"
            >
              <option value=""></option>
              <option
                *ngFor="let opt of lookupOptions()[field.lookupCode || '']"
                [value]="opt"
              >{{ opt }}</option>
            </select>
            <div
              class="fsa-error-message"
              *ngIf="form.get(field.key)?.errors?.['required'] && form.get(field.key)?.touched"
            >
              Required
            </div>
          </ng-container>

          <!-- LABEL type (column headers) -->
          <ng-container *ngIf="field.type === 'label' && field.key !== '__spacer__'">
            <label class="fsa-field__label">{{ field.label }}</label>
          </ng-container>

        </div>

      </ng-container>
    </ng-container>

    <!-- Action Buttons -->
    <div class="fsa-grid__1">
      <button
        type="button"
        class="fsa-btn fsa-btn--secondary"
        (click)="onCancel()"
      >Cancel</button>
      <button
        type="button"
        class="fsa-btn fsa-btn--primary"
        (click)="onSubmit()"
      >Save</button>
    </div>

  </div>
</div>

<ng-template #loadingTemplate>
  <div class="fsa-loading" aria-live="polite" aria-busy="true">
    <span class="fsa-sr-only">Loading&hellip;</span>
