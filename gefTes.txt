import { TestBed } from '@angular/core/testing';
import { FormBuilder, Validators } from '@angular/forms';
import { YourComponent } from './your.component'; // adjust path

describe('YourComponent - buildForm()', () => {
  let component: YourComponent;
  let fb: FormBuilder;

  beforeEach(() => {
    TestBed.configureTestingModule({
      providers: [FormBuilder],
      declarations: [YourComponent], // or imports if standalone
    });

    const fixture = TestBed.createComponent(YourComponent);
    component = fixture.componentInstance;
    fb = TestBed.inject(FormBuilder);
  });

  // ─── yr_select ───────────────────────────────────────────

  it('should set yr_select_0 to true when first year is selected', () => {
    component.programYears = [{ selected: true, status: 'A' }];
    component.buildForm();
    expect(component.form.get('yr_select_0')?.value).toBe(true);
  });

  it('should set yr_select_0 to false when first year is not selected', () => {
    component.programYears = [{ selected: false, status: 'A' }];
    component.buildForm();
    expect(component.form.get('yr_select_0')?.value).toBe(false);
  });

  // ─── yr_status: y.status truthy branch ──────────────────

  it('should use y.status when status is "A"', () => {
    component.programYears = [{ selected: true, status: 'A' }];
    component.buildForm();
    expect(component.form.get('yr_status_0')?.value).toBe('A');
  });

  it('should use y.status when status is "I"', () => {
    component.programYears = [{ selected: false, status: 'I' }];
    component.buildForm();
    expect(component.form.get('yr_status_0')?.value).toBe('I');
  });

  // ─── yr_status: y.status falsy branch (|| 'I') ──────────

  it('should default status to "I" when y.status is null', () => {
    component.programYears = [{ selected: false, status: null }];
    component.buildForm();
    expect(component.form.get('yr_status_0')?.value).toBe('I');
  });

  it('should default status to "I" when y.status is undefined', () => {
    component.programYears = [{ selected: false, status: undefined }];
    component.buildForm();
    expect(component.form.get('yr_status_0')?.value).toBe('I');
  });

  it('should default status to "I" when y.status is empty string', () => {
    component.programYears = [{ selected: false, status: '' }];
    component.buildForm();
    expect(component.form.get('yr_status_0')?.value).toBe('I');
  });

  // ─── yr_status validators: y.selected true branch ───────

  it('should apply required and pattern validators when selected is true', () => {
    component.programYears = [{ selected: true, status: 'A' }];
    component.buildForm();

    const control = component.form.get('yr_status_0');

    // Valid value passes
    control?.setValue('A');
    expect(control?.valid).toBe(true);

    control?.setValue('I');
    expect(control?.valid).toBe(true);

    // Required: empty fails
    control?.setValue('');
    expect(control?.hasError('required')).toBe(true);

    // Pattern: invalid value fails
    control?.setValue('X');
    expect(control?.hasError('pattern')).toBe(true);
  });

  // ─── yr_status validators: y.selected false branch ──────

  it('should apply no validators when selected is false', () => {
    component.programYears = [{ selected: false, status: 'I' }];
    component.buildForm();

    const control = component.form.get('yr_status_0');

    // No validators — any value should be valid
    control?.setValue('');
    expect(control?.valid).toBe(true);

    control?.setValue('X');
    expect(control?.valid).toBe(true);
  });

  // ─── Multiple years ──────────────────────────────────────

  it('should build controls for all programYears with correct indexes', () => {
    component.programYears = [
      { selected: true,  status: 'A' },
      { selected: false, status: ''  },
      { selected: true,  status: null },
    ];
    component.buildForm();

    expect(component.form.get('yr_select_0')?.value).toBe(true);
    expect(component.form.get('yr_select_1')?.value).toBe(false);
    expect(component.form.get('yr_select_2')?.value).toBe(true);

    expect(component.form.get('yr_status_0')?.value).toBe('A');
    expect(component.form.get('yr_status_1')?.value).toBe('I'); // fallback
    expect(component.form.get('yr_status_2')?.value).toBe('I'); // null fallback
  });

  it('should build an empty form group when programYears is empty', () => {
    component.programYears = [];
    component.buildForm();
    expect(Object.keys(component.form.controls).length).toBe(0);
  });
});
