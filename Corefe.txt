@Override
public Page<CommodityBO> findAll(Pageable pageable) {

    // List of commodity codes considered "standard"
    Set<Integer> standardCodes = Set.of(
        1000, 2000, 2200, 2300, 2500, 2600, 2700, 4165, 4210,
        4600, 4850, 4860, 5100, 6000, 6100, 6300, 6600, 6900,
        7100, 7200, 7300, 7400, 7600, 7700, 7800, 8000, 8204,
        8301, 8305, 8306, 8311, 9500
    );

    return commodityDao.findAll(pageable)
        .map(entity -> {
            CommodityBO bo = modelMapper.map(entity, CommodityBO.class);

            // Set isStandard based on whether commodityCode is found in the list
            boolean isStandard = standardCodes.contains(entity.getCommodityCode());
            bo.setIsStandard(isStandard);

            return bo;
        });
}
