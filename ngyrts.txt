import { Component, OnInit } from ‘@angular/core’;
import { CommonModule } from ‘@angular/common’;
import { FormsModule } from ‘@angular/forms’;
import { ApiService } from ‘../../../services/api.service’;
import { HttpErrorResponse } from ‘@angular/common/http’;

interface AccountingProgramYear {
year: number;
status: ‘Active’ | ‘Inactive’;
selected: boolean;
}

@Component({
selector: ‘app-accounting-program-years’,
standalone: true,
imports: [CommonModule, FormsModule],
templateUrl: ‘./accounting-program-years.component.html’,
styleUrls: [’./accounting-program-years.component.scss’]
})
export class AccountingProgramYearsComponent implements OnInit {
errorMessage: string = ‘’;
saveSubmitResultMessage: string = ‘’;

programYears: AccountingProgramYear[] = [];
isLoadingYears: boolean = false;

constructor(private apiService: ApiService) {}

ngOnInit(): void {
this.initializeYears();
}

initializeYears(): void {
const currentYear = new Date().getFullYear();
this.programYears = [];

```
// Generate current year + 9 consecutive years (total 10 years)
for (let i = 0; i < 10; i++) {
  this.programYears.push({
    year: currentYear + i,
    status: 'Inactive',
    selected: false
  });
}
```

}

onSaveDraft(): void {
const selectedYears = this.captureSelectedYears();
console.log(‘selectedYears’, selectedYears);

```
if (selectedYears.length === 0) {
  this.errorMessage = 'Please select at least one year';
  return;
}

this.errorMessage = '';

const draftJson = {
  programYears: selectedYears,
  status: 'DRAFT'
};

console.log('draftJson', draftJson);

this.apiService.post('pfrsuiapi/AccountingProgramYears/saveDraft', draftJson, { responseType: 'text' }).subscribe({
  next: (res) => {
    this.saveSubmitResultMessage = 'Draft saved successfully';
    console.log('response from save draft API:', res);

    // Show the save submit operation result only for 5 seconds
    setTimeout(() => {
      this.saveSubmitResultMessage = '';
    }, 5000);
  },
  error: (err: HttpErrorResponse) => {
    this.errorMessage = 'Error saving accounting program years draft';
    console.error('Error saving accounting program years draft', err);
  }
});
```

}

onSubmit(): void {
const selectedYears = this.captureSelectedYears();

```
if (selectedYears.length === 0) {
  this.errorMessage = 'Please select at least one year';
  return;
}

this.errorMessage = '';
console.log('Submitting selected years:', selectedYears);

const submitJson = {
  programYears: selectedYears,
  status: 'SUBMITTED'
};

this.apiService.post('pfrsuiapi/AccountingProgramYears/submit', submitJson, { responseType: 'text' }).subscribe({
  next: (res) => {
    this.saveSubmitResultMessage = 'Years submitted successfully';
    console.log('response from submit API:', res);

    setTimeout(() => {
      this.saveSubmitResultMessage = '';
    }, 5000);
  },
  error: (err: HttpErrorResponse) => {
    this.errorMessage = 'Error submitting accounting program years';
    console.error('Error submitting accounting program years', err);
  }
});
```

}

captureSelectedYears(): AccountingProgramYear[] {
return this.programYears.filter(year => year.selected);
}

toggleYearSelection(year: AccountingProgramYear): void {
year.selected = !year.selected;
}

selectAllYears(): void {
this.programYears.forEach(year => year.selected = true);
}

deselectAllYears(): void {
this.programYears.forEach(year => year.selected = false);
}

getSelectedCount(): number {
return this.programYears.filter(year => year.selected).length;
}

isAnyYearSelected(): boolean {
return this.programYears.some(year => year.selected);
}
}
