import React, { useState, useEffect } from 'react';
import { GridComponent, ColumnsDirective, ColumnDirective, Page, Inject } from '@syncfusion/ej2-react-grids';
import { SpinnerComponent } from '@syncfusion/ej2-react-popups';
import '@syncfusion/ej2-react-grids/styles/material.css';
import '@syncfusion/ej2-react-popups/styles/material.css';

const App: React.FC = () => {
  const [loading, setLoading] = useState<boolean>(false);
  const [data, setData] = useState<any[]>([]);

  const fetchData = async () => {
    setLoading(true);
    // Simulate an API call
    setTimeout(() => {
      setData([
        { OrderID: 10248, CustomerName: 'Paul Henriot', Freight: 32.38, OrderDate: new Date(8364186e5) },
        { OrderID: 10249, CustomerName: 'Karin Josephs', Freight: 11.61, OrderDate: new Date(836505e6) },
        // Add more data here
      ]);
      setLoading(false);
    }, 2000);
  };

  useEffect(() => {
    fetchData();
  }, []);

  const handleButtonClick = () => {
    fetchData();
  };

  return (
    <div>
      {loading && <SpinnerComponent visible={true} />}
      <button onClick={handleButtonClick} disabled={loading}>
        Refresh Data
      </button>
      <GridComponent dataSource={data} allowPaging={true}>
        <ColumnsDirective>
          <ColumnDirective field='OrderID' headerText='Order ID' width='100' textAlign='Right'/>
          <ColumnDirective field='CustomerName' headerText='Customer Name' width='150'/>
          <ColumnDirective field='Freight' headerText='Freight' width='100' format='C2' textAlign='Right'/>
          <ColumnDirective field='OrderDate' headerText='Order Date' width='130' format='yMd' textAlign='Right'/>
        </ColumnsDirective>
        <Inject services={[Page]} />
      </GridComponent>
    </div>
  );
};

export default App;
