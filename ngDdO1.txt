import { Injectable } from '@angular/core';
import { HttpClient } from '@angular/common/http';
import { Observable } from 'rxjs';

export interface CmdtyOtherOption {
  code: string;
  description: string;
}

@Injectable({ providedIn: 'root' })
export class CommodityService {
  // TODO: change URL to match your backend
  private readonly baseUrl = '/api/commodities';

  constructor(private http: HttpClient) {}

  getOtherCommodityOptions(): Observable<CmdtyOtherOption[]> {
    return this.http.get<CmdtyOtherOption[]>(`${this.baseUrl}/other`);
  }
}








import { Component, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { FormsModule } from '@angular/forms';
import { CommodityService, CmdtyOtherOption } from '../services/commodity.service';

@Component({
  selector: 'app-commodity',
  standalone: true,
  imports: [CommonModule, FormsModule],
  templateUrl: './commodity.component.html',
  styleUrls: ['./commodity.component.scss']
})
export class CommodityComponent implements OnInit {
  pageTitle = 'Commodities';
  errorMessage = '';

  // "Other" dropdown options from API
  cmdtyOtherOptions: CmdtyOtherOption[] = [];

  // selected value from the "Other" dropdown
  cmdtyOtherCode = '';

  // ... existing commodityOptions, commoditySelection, etc. stay the same ...

  constructor(private commodityService: CommodityService) {
    this.commodityOptions.forEach(o => (this.commoditySelection[o.id] = false));
  }

  ngOnInit(): void {
    this.loadOtherCommodityOptions();
  }

  private loadOtherCommodityOptions(): void {
    this.commodityService.getOtherCommodityOptions().subscribe({
      next: (options) => (this.cmdtyOtherOptions = options),
      error: (err) => {
        console.error('Error loading Other commodities', err);
        this.errorMessage = 'Unable to load Other commodities.';
      }
    });
  }

  reset(): void {
    Object.keys(this.commoditySelection).forEach(
      key => (this.commoditySelection[key] = false)
    );
    this.cmdtyOtherCode = '';
    this.errorMessage = '';
  }

  private buildPayload(): {
    selectedCommodities: any[];
    otherCommodity: CmdtyOtherOption | null;
  } {
    const selectedCommodities = this.commodityOptions.filter(
      o => this.commoditySelection[o.id]
    );

    const otherCommodity =
      this.cmdtyOtherOptions.find(o => o.code === this.cmdtyOtherCode) ?? null;

    return { selectedCommodities, otherCommodity };
  }

  // onSaveDraft, onSubmit, selectAll stay the same except they now use buildPayload()
}
