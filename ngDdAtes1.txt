  expect(component.errorMessage).toBe('Please select at least one commodity');
});

  component.onSubmit();

  component.cmdtyWheat1000 = true;
  component.cmdtyCorn2000 = true;
  component.cmdtyOther = '1';

  component.onSubmit();
  
  expect(component.errorMessage).toBe('');

});
