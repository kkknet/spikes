import static org.junit.jupiter.api.Assertions.*;
import static org.mockito.Mockito.*;

import java.util.List;
import java.util.Map;

import org.junit.jupiter.api.Test;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.validation.BindingResult;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;

class GlobalExceptionHandlerTest {

    private final GlobalExceptionHandler handler = new GlobalExceptiontionHandler();

    @Test
    void handleValidationExceptions_shouldReturnBadRequestWithFieldErrorsMap() {
        // Arrange
        MethodArgumentNotValidException ex = mock(MethodArgumentNotValidException.class);
        BindingResult bindingResult = mock(BindingResult.class);

        List<FieldError> fieldErrors = List.of(
                new FieldError("obj", "firstName", "First name is required"),
                new FieldError("obj", "age", "Age must be >= 18")
        );

        when(ex.getBindingResult()).thenReturn(bindingResult);
        when(bindingResult.getFieldErrors()).thenReturn(fieldErrors);

        // Act
        ResponseEntity<Map<String, String>> response = handler.handleValidationExceptions(ex);

        // Assert
        assertEquals(HttpStatus.BAD_REQUEST, response.getStatusCode());
        assertNotNull(response.getBody());
        assertEquals(2, response.getBody().size());
        assertEquals("First name is required", response.getBody().get("firstName"));
        assertEquals("Age must be >= 18", response.getBody().get("age"));
    }

    @Test
    void handleValidationExceptions_shouldReturnBadRequestWithEmptyMapWhenNoErrors() {
        // Arrange
        MethodArgumentNotValidException ex = mock(MethodArgumentNotValidException.class);
        BindingResult bindingResult = mock(BindingResult.class);

        when(ex.getBindingResult()).thenReturn(bindingResult);
        when(bindingResult.getFieldErrors()).thenReturn(List.of());

        // Act
        ResponseEntity<Map<String, String>> response = handler.handleValidationExceptions(ex);

        // Assert
        assertEquals(HttpStatus.BAD_REQUEST, response.getStatusCode());
        assertNotNull(response.getBody());
        assertTrue(response.getBody().isEmpty());
    }

    @Test
    void handlePfrsUiDataAccessException_shouldReturnInternalServerErrorWithMessage() {
        // Arrange
        Throwable t = new RuntimeException("DB connection failed");

        // Act
        ResponseEntity<Object> response = handler.handlePfrsUiDataAccessException(t);

        // Assert
        assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, response.getStatusCode());
        assertEquals("DB connection failed", response.getBody());
    }

    @Test
    void handlePfrsUiDataAccessException_shouldReturnInternalServerErrorWithNullBodyIfMessageNull() {
        // Arrange
        Throwable t = new RuntimeException((String) null);

        // Act
        ResponseEntity<Object> response = handler.handlePfrsUiDataAccessException(t);

        // Assert
        assertEquals(HttpStatus.INTERNAL_SERVER_ERROR, response.getStatusCode());
        assertNull(response.getBody());
    }
}
